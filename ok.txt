# --- DATABASE SETUP ---
def init_db():
    conn = sqlite3.connect(DB_FILE)
    c = conn.cursor()
    c.execute("""
        CREATE TABLE IF NOT EXISTS requests (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            teacher_name TEXT NOT NULL,
            school_name TEXT NOT NULL,
            district TEXT NOT NULL,
            title TEXT NOT NULL,
            category TEXT NOT NULL,
            description TEXT,
            goal_amount REAL NOT NULL,
            raised_amount REAL DEFAULT 0.0,
            created_at TEXT
        )
    """)
    c.execute("""
        CREATE TABLE IF NOT EXISTS donations (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            request_id INTEGER,
            donor_name TEXT,
            amount REAL,
            message TEXT,
            stripe_session_id TEXT UNIQUE,
            donated_at TEXT
        )
    """)
    c.execute("SELECT COUNT(*) FROM requests")
    if c.fetchone() == 0:
        samples = [
            ("Ms. Alvarez", "Lakeridge Middle School", "Sumner-Bonney Lake", "STEM Robotics & Sensor Kits", "STEM & Tech", "Programmable microcontrollers and ultrasonic sensors for 8th-grade engineering.", 450.0, 180.0, datetime.now().strftime("%Y-%m-%d")),
            ("Mr. Jenkins", "Bonney Lake High School", "Sumner-Bonney Lake", "AP Physics Lab Equipment", "Science", "Digital force sensors and dynamics carts for mechanics labs.", 600.0, 425.0, datetime.now().strftime("%Y-%m-%d")),
            ("Mrs. Chen", "Emerald Ridge High School", "Puyallup", "Classroom Library & Novels", "Books & Literacy", "Diverse contemporary fiction and non-fiction sets for 10th-grade English.", 300.0, 90.0, datetime.now().strftime("%Y-%m-%d")),
        ]
        c.executemany("""
            INSERT INTO requests (teacher_name, school_name, district, title, category, description, goal_amount, raised_amount, created_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, samples)
    conn.commit()
    conn.close()

init_db()

# --- VERIFY PAYMENT WHEN RETURNING FROM STRIPE ---
if "session_id" in st.query_params:
    sid = st.query_params["session_id"]
    conn = sqlite3.connect(DB_FILE)
    c = conn.cursor()
    c.execute("SELECT id FROM donations WHERE stripe_session_id = ?", (sid,))
    if not c.fetchone():
        try:
            session = stripe.checkout.Session.retrieve(sid)
            if session.payment_status == "paid":
                req_id = int(session.metadata["request_id"])
                donor = session.metadata.get("donor_name", "Anonymous")
                msg = session.metadata.get("message", "")
                dollars = session.amount_total / 100.0
                c.execute("INSERT INTO donations (request_id, donor_name, amount, message, stripe_session_id, donated_at) VALUES (?, ?, ?, ?, ?, ?)",
                          (req_id, donor, dollars, msg, sid, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
                c.execute("UPDATE requests SET raised_amount = raised_amount + ? WHERE id = ?", (dollars, req_id))
                conn.commit()
                st.success(f"Thank you, {donor}! Your ${dollars:,.2f} donation is confirmed.")
                st.balloons()
        except Exception as e:
            st.error(f"Payment verification error: {e}")
    conn.close()
    st.query_params.clear()

# --- MAIN APP UI ---
st.sidebar.title("Navigation")
page = st.sidebar.radio("Go to", ["Browse Classrooms", "Teacher Request Portal"])

if page == "Browse Classrooms":
    st.title("📚 Pierce County Classroom Donations")
    st.write("Directly fund classroom supplies and STEM projects for local teachers.")

    conn = sqlite3.connect(DB_FILE)
    df = pd.read_sql_query("SELECT * FROM requests ORDER BY id DESC", conn)
    conn.close()

    for _, row in df.iterrows():
        with st.container(border=True):
            c1, c2 = st.columns(2)
            with c1:
                st.subheader(row["title"])
                st.caption(f"**{row['teacher_name']}** • {row['school_name']} ({row['district']})")
                st.write(row["description"])
                pct = min(row["raised_amount"] / row["goal_amount"], 1.0) if row["goal_amount"] > 0 else 0.0
                st.progress(pct)
                st.write(f"**${row['raised_amount']:,.2f}** raised of **${row['goal_amount']:,.2f}** goal")

            with c2:
                donor = st.text_input("Your Name", placeholder="Anonymous", key=f"name_{row['id']}")
                amt = st.number_input("Amount ($)", min_value=5.0, value=25.0, step=5.0, key=f"amt_{row['id']}")
                msg = st.text_input("Note to Teacher", placeholder="Great project!", key=f"msg_{row['id']}")
                
                if st.button("Proceed to Checkout", key=f"btn_{row['id']}", use_container_width=True):
                    try:
                        session = stripe.checkout.Session.create(
                            payment_method_types=["card"],
                            line_items=[{
                                "price_data": {
                                    "currency": "usd",
                                    "product_data": {"name": f"Donation: {row['title']}"},
                                    "unit_amount": int(amt * 100),
                                },
                                "quantity": 1,
                            }],
                            mode="payment",
                            metadata={"request_id": str(row["id"]), "donor_name": donor or "Anonymous", "message": msg},
                            success_url=f"{APP_URL}/?session_id={{CHECKOUT_SESSION_ID}}",
                            cancel_url=f"{APP_URL}/",
                        )
                        st.link_button("💳 Pay Now on Stripe", session.url, type="primary", use_container_width=True)
                    except Exception as e:
                        st.error(f"Stripe Error: {e}")

                      (t_name, s_name, dist, p_title, cat, desc, goal, datetime.now().strftime("%Y-%m-%d")))

            st.success("Published!")
